# 1. Initialize Git in this folder
git init

# 2. Get the status
git status

# 3. Stage all your files
git add .

# 4. Commit them with a message
git commit -m "Initial commit"

# 5. Link your local repo to GitHub
git remote add origin https://github.com/yourusername/mydevrepo.git

# 6. Push to GitHub
git branch -M main
git push -u origin main
